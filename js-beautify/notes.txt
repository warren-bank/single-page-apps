bash:
=====

zip_file='release.zip'
src_dir='js-beautify-release'

wget -O "$zip_file" 'https://github.com/beautifier/js-beautify/archive/refs/heads/release.zip'
unzip "$zip_file"

mv "${src_dir}/index.html" .
mkdir 'js'
mv "${src_dir}/js/lib" ./js
mv "${src_dir}/web" .

rm -f "$zip_file"
rm -rf "$src_dir"
